function albHistRender(){
  const albaranes=JSON.parse(localStorage.getItem('colch_albaranes_recibidos')||'[]');
  const filtroFecha=document.getElementById('albhist-filtro-fecha')?.value||'';
  const filtroProv=document.getElementById('albhist-filtro-prov')?.value||'';
  const filtroRef=(document.getElementById('albhist-filtro-ref')?.value||'').trim().toLowerCase();

  const normFecha=f=>{
    if(!f) return '';
    if(f.includes('-')) return f;
    const[d,m,y]=f.split('/');
    return `${y}-${m.padStart(2,'0')}-${d.padStart(2,'0')}`;
  };

  const filtrados=albaranes.filter(a=>{
    if(filtroFecha && normFecha(a.fecha)!==filtroFecha) return false;
    if(filtroProv && a.proveedor!==filtroProv) return false;
    if(filtroRef && !(a.ref||'').toLowerCase().includes(filtroRef)) return false;
    return true;
  });

  const stats=document.getElementById('albhist-stats');
  const content=document.getElementById('albhist-content');

  const totalUds=filtrados.reduce((sum,a)=>{
    return sum+(a.lineas||[]).reduce((s,l)=>s+(parseInt(l.cantidad)||0),0);
  },0);

  stats.innerHTML=`
    
Albaranes
${filtrados.length}

    
Unidades totales
${totalUds}

    
Total guardados
${albaranes.length}
`;

  if(!filtrados.length){
    content.innerHTML='
No hay albaranes que coincidan con los filtros.
';
    return;
  }

  content.innerHTML=filtrados.map((a,idx)=>{
    const provColor=a.proveedor==='NEW MATTRESS'?'#3730A3':'#3B4A2F';
    const provBg=a.proveedor==='NEW MATTRESS'?'#EEF2FF':'#F4F6F1';
    const provLabel=a.proveedor==='NEW MATTRESS'?'NEW MATTRESS':a.proveedor==='LUSOCOLCHAO'?'LUSOCOLCHÃO':a.proveedor||'—';
    const totalUdsAlb=(a.lineas||[]).reduce((s,l)=>s+(parseInt(l.cantidad)||0),0);

    // ── Tabla de líneas con detalle de pedidos asociados ──
    const lineasSrc=a.detalleLineas||((a.lineas||[]).map(l=>({...l,pedidos:[],sinPedido:0})));
    const lineasHTML=lineasSrc.map(l=>{
      const pedidosHTML=(l.pedidos||[]).length>0
        ? (l.pedidos||[]).map(p=>`${p.pedidoId}${p.fechaPedido?' · '+p.fechaPedido:''} · ${p.uds}ud`).join('')
        : '';
      const sinPedidoHTML=l.sinPedido>0
        ? `${l.sinPedido}ud sin pedido`
        : '';
      return `
        ${l.modelo||'—'}
        ${l.medida||'—'}
        ${l.cantidad||0}
        ${pedidosHTML}${sinPedidoHTML}${!pedidosHTML&&!sinPedidoHTML?'—':''}
      `;
    }).join('');

    // ── Visor PDF (si hay base64) ──
    const pdfViewerId=`albpdf-${idx}`;
    const pdfBtnHTML=a.pdfB64
      ? ``
      : `Sin PDF adjunto`;

    const pdfViewerHTML=a.pdfB64
      ? `

          

             Documento PDF
            
          

          
        
`
      : '';

    return `

      
      

        

          ${provLabel}
          ${a.ref?'Alb. '+a.ref:'Sin nº albarán'}
          ${a.fecha||'Sin fecha'}
        

        

          ${totalUdsAlb} ud${totalUdsAlb!==1?'s':''}
          ${pdfBtnHTML}
        

      

      
      

        ${lineasHTML}

          
            

              Modelo
              Medida
              Uds
              Pedido asociado · fecha
            

          
          
        

      

      
      

        ${pdfViewerHTML}
      

    
`;
  }).join('');
}

function albHistVerPDF(id){
  const el=document.getElementById(id);
  if(!el) return;
  el.style.display=el.style.display==='none'?'block':'none';
}

function albHistLimpiarFiltros(){
  const f=document.getElementById('albhist-filtro-fecha');
  const p=document.getElementById('albhist-filtro-prov');
  const r=document.getElementById('albhist-filtro-ref');
  if(f) f.value='';
  if(p) p.value='';
  if(r) r.value='';
  albHistRender();
}
